import { ColorItem, ColorPalette, Meta } from '@storybook/addon-docs/blocks'
import { cssVar, defaultTheme, flattenSemantic, intents, paletteNames } from '@octarine/tokens'

<Meta title="Foundations/Colors" />

export const palettes = paletteNames.map((name) => [name, defaultTheme.palette[name]])

export const semantic = flattenSemantic(defaultTheme.light.colors).map(([name]) => name)

export const swatches = (names) =>
  Object.fromEntries(names.map((name) => [name, `var(${cssVar(`color-${name}`)})`]))

# Colors

Octarine colors come in two layers, both defined in `@octarine/tokens`:

- **Primitives** — raw OKLCH scales (`brand-500`, `neutral-200`, …). They carry no meaning.
- **Semantic** — named by purpose (`surface`, `fg-muted`, `danger`, …) and pointing at a primitive.
  They swap between light and dark: use the **Mode** toolbar button to compare.

Each theme brings its own palettes and mappings; switch **Theme** in the toolbar to see them.

Components only use semantic colors, so `bg-primary` or `text-fg-muted` adapt to the theme
automatically. Reach for a primitive (`bg-brand-100`) only for one-off illustrations.

## Semantic

<ColorPalette>
  <ColorItem
    title="Surfaces"
    subtitle="bg-canvas, bg-surface, …"
    colors={swatches(['canvas', 'surface', 'surface-raised', 'surface-sunken'])}
  />
  <ColorItem
    title="Foreground"
    subtitle="text-fg, text-fg-muted, …"
    colors={swatches(['fg', 'fg-muted', 'fg-subtle', 'fg-inverse'])}
  />
  <ColorItem
    title="Lines"
    subtitle="border-border, ring"
    colors={swatches(['border', 'border-strong', 'ring'])}
  />
  {intents.map((intent) => (
    <ColorItem
      key={intent}
      title={intent}
      subtitle={`bg-${intent}, bg-${intent}-subtle, …`}
      colors={swatches(semantic.filter((name) => name === intent || name.startsWith(`${intent}-`)))}
    />
  ))}
</ColorPalette>

## Primitives
